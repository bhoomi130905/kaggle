PRACTICAL 1: Implementation of 16-Qubit Random Number Generator

NOTE:
Uses local Qiskit Aer simulation; no IBM Quantum account/API key is required. A simulator is not a physical quantum entropy source.

# If needed, install dependencies once in your terminal:
# python -m pip install qiskit qiskit-aer matplotlib

from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator

NUM_QUBITS = 16
SHOTS = 1

circuit = QuantumCircuit(NUM_QUBITS, NUM_QUBITS)
circuit.h(range(NUM_QUBITS))                 # Put all 16 qubits into superposition
circuit.measure(range(NUM_QUBITS), range(NUM_QUBITS))

print("16-qubit random number generator circuit:")
print(circuit.draw())

simulator = AerSimulator()
compiled_circuit = transpile(circuit, simulator)
result = simulator.run(compiled_circuit, shots=SHOTS).result()
counts = result.get_counts()

binary_number = next(iter(counts))
decimal_number = int(binary_number.replace(" ", ""), 2)

print("Random number (binary):", binary_number)
print("Random number (decimal):", decimal_number)
